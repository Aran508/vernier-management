Option Explicit

Private Sub Workbook_Open()
    On Error Resume Next
    modMain.SystemStartup
    On Error GoTo 0
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    modAutoCAD.ReleaseAcad
    modOutlook.ReleaseOutlook
    Application.StatusBar = False
    On Error GoTo 0
End Sub

Private Sub Workbook_SheetActivate(ByVal Sh As Object)
    On Error Resume Next
    If Sh.Name = SH_CONTROL Then modDashboard.RefreshDashboard
    On Error GoTo 0
End Sub
